function unlockAnimation(zone){ completedLocks++; const overlay = document.getElementById('vault-overlay'); overlay.style.display = 'flex'; const lock = document.querySelector(`.lock${zone + 1}`); setTimeout(() => { lock.classList.add('removed'); }, 500); document.getElementById('vault-text').innerText = `${4 - completedLocks} LOCKS REMAIN`; if(completedLocks < 4){ setTimeout(() => { overlay.style.display = 'none'; }, 3500); } else { setTimeout(() => { document.getElementById('vault-text').innerText = 'ALL LOCKS RELEASED'; document.getElementById('vault-door') .classList.add('open'); setTimeout(() => { document.getElementById('final') .style.display = 'flex'; setInterval(firework, 400); }, 3500); }, 2500); } } function firework(){ const f = document.createElement('div'); f.className = 'firework'; f.style.left = Math.random() * 100 + 'vw'; f.style.top = Math.random() * 80 + 'vh'; f.style.background = `hsl(${Math.random() * 360},100%,50%)`; document.getElementById('final').appendChild(f); setTimeout(() => { f.remove(); }, 1000); }
